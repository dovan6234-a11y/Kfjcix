	local dark = Color3.fromRGB(14, 14, 14)
	local g = new("UIGradient", {
		Color = ColorSequence.new({
			ColorSequenceKeypoint.new(0, dark),

	return g

	local char = player.Character
	local hum = char and char:FindFirstChildOfClass("Humanoid")
	local root = char and char:FindFirstChild("HumanoidRootPart")
	if not hum or not root then return end

		hum.UseJumpPower = true
		hum.JumpPower = State.Jump

	if State.SpinOn then
		hum.AutoRotate = false
		root.CFrame = root.CFrame * CFrame.Angles(0, math.rad(State.Spin * 10) * dt, 0)
	elseif wasSpin then
		hum.AutoRotate = true
	end
	wasSpin = State.SpinOn
end)

local KNIFE_NAME = "Knife"
local GUN_NAME = "Gun"

local function getRole(plr)
	local char = plr.Character
	local bp = plr:FindFirstChildOfClass("Backpack")
	if (char and char:FindFirstChild(KNIFE_NAME)) or (bp and bp:FindFirstChild(KNIFE_NAME)) then
		return "Murderer"
	end
	if (char and char:FindFirstChild(GUN_NAME)) or (bp and bp:FindFirstChild(GUN_NAME)) then
		return "Sheriff"
	end
	return "Innocent"
end

local function getMurderer()
	for _, plr in Players:GetPlayers() do
		local char = plr.Character
		if plr ~= player and char and getRole(plr) == "Murderer" then
			local hum = char:FindFirstChildOfClass("Humanoid")
			if hum and hum.Health > 0 then return plr end
		end
	end
end

local rayParams = RaycastParams.new()
rayParams.FilterType = Enum.RaycastFilterType.Exclude

local function getAimPart(char)
	if State.AimPart == "Head" then
		return char:FindFirstChild("Head")
	elseif State.AimPart == "Torso" then
		return char:FindFirstChild("UpperTorso") or char:FindFirstChild("Torso")
	else
		return char:FindFirstChild("LeftLowerLeg") or char:FindFirstChild("Left Leg")
	end
end

local function isVisible(cam, part, char)
	rayParams.FilterDescendantsInstances = {player.Character}
	local origin = cam.CFrame.Position
	local result = Workspace:Raycast(origin, part.Position - origin, rayParams)
	return result == nil or result.Instance:IsDescendantOf(char)
end

local function getTarget(cam)
	local center = cam.ViewportSize / 2
	local best, bestDist = nil, State.FOV
	for _, plr in Players:GetPlayers() do
		local char = plr.Character
		if plr ~= player and char then
			local hum = char:FindFirstChildOfClass("Humanoid")
			local part = getAimPart(char)
			if hum and hum.Health > 0 and part then
				local pos, onScreen = cam:WorldToViewportPoint(part.Position)
				if onScreen then
					local d = (Vector2.new(pos.X, pos.Y) - center).Magnitude
					if d < bestDist and (not State.WallCheck or isVisible(cam, part, char)) then
						best, bestDist = part, d
					end
				end
			end
		end
	end
	return best
end

RunService:BindToRenderStep("AimBot", Enum.RenderPriority.Camera.Value + 1, function(dt)
	local cam = Workspace.CurrentCamera
	if not cam then return end

	local vs = cam.ViewportSize
	fovCircle.Position = UDim2.fromOffset(vs.X / 2, vs.Y / 2)
	fovCircle.Size = UDim2.fromOffset(State.FOV * 2, State.FOV * 2)
	fovCircle.Visible = State.FOVCircle

	if State.CamFrozen or not (State.Aimbot or State.LockMurderer) then return end
	if State.HoldRMB and not UIS:IsMouseButtonPressed(Enum.UserInputType.MouseButton2) then return end

	local target
	if State.LockMurderer then
		local m = getMurderer()
		if m then target = getAimPart(m.Character) end
	end
	if not target and State.Aimbot then
		target = getTarget(cam)
	end
	if target then
		local cf = cam.CFrame
		local goal = CFrame.lookAt(cf.Position, target.Position)
		cam.CFrame = cf:Lerp(goal, math.clamp(dt * 60 / State.Smoothing, 0, 1))
	end
end)

local espGui = new("ScreenGui", {
	Name = "AstralESP", ResetOnSpawn = false, DisplayOrder = 5, IgnoreGuiInset = true,
}, playerGui)

local espData = {}

local function destroyESP(plr)
	local d = espData[plr]
	if d then
		d.box:Destroy()
		d.text:Destroy()
		d.highlight:Destroy()
		d.line:Destroy()
		for _, f in d.bones do f:Destroy() end
		espData[plr] = nil
	end
end

local function createESP(plr, char, root, head)
	local box = new("BillboardGui", {
		Adornee = root, AlwaysOnTop = true, Size = UDim2.fromScale(4, 6),
		ResetOnSpawn = false,
	}, espGui)
	local boxFrame = new("Frame", {Size = UDim2.fromScale(1, 1), BackgroundTransparency = 1}, box)
	local stroke = new("UIStroke", {Thickness = 1.5, Color = WHITE}, boxFrame)

	local text = new("BillboardGui", {
		Adornee = head, AlwaysOnTop = true, Size = UDim2.fromOffset(200, 36),
		StudsOffset = Vector3.new(0, 2.6, 0), ResetOnSpawn = false,
	}, espGui)
	local label = new("TextLabel", {
		Size = UDim2.fromScale(1, 1), BackgroundTransparency = 1, TextColor3 = WHITE,
		TextStrokeTransparency = 0, Font = Enum.Font.GothamBold, TextSize = 13,
		TextYAlignment = Enum.TextYAlignment.Bottom,
	}, text)

	local highlight = new("Highlight", {
		FillTransparency = 0.5, OutlineTransparency = 0,
		DepthMode = Enum.HighlightDepthMode.AlwaysOnTop,
	}, char)

	local line = new("Frame", {
		AnchorPoint = Vector2.new(0.5, 0.5), BorderSizePixel = 0,
		BackgroundColor3 = WHITE, Visible = false,
	}, espGui)

	espData[plr] = {
		char = char, box = box, stroke = stroke, text = text, label = label,
		highlight = highlight, line = line, bones = {}, boneMax = 0,
	}
	return espData[plr]
end

local BONES_R15 = {
	{"Head", "UpperTorso"}, {"UpperTorso", "LowerTorso"},
	{"UpperTorso", "LeftUpperArm"}, {"LeftUpperArm", "LeftLowerArm"}, {"LeftLowerArm", "LeftHand"},
	{"UpperTorso", "RightUpperArm"}, {"RightUpperArm", "RightLowerArm"}, {"RightLowerArm", "RightHand"},
	{"LowerTorso", "LeftUpperLeg"}, {"LeftUpperLeg", "LeftLowerLeg"}, {"LeftLowerLeg", "LeftFoot"},
	{"LowerTorso", "RightUpperLeg"}, {"RightUpperLeg", "RightLowerLeg"}, {"RightLowerLeg", "RightFoot"},
}
local BONES_R6 = {
	{"Head", "Torso"}, {"Torso", "Left Arm"}, {"Torso", "Right Arm"},
	{"Torso", "Left Leg"}, {"Torso", "Right Leg"},
}

local function drawLine(frame, from, to, color, thickness)
	local diff = to - from
	frame.Size = UDim2.fromOffset(diff.Magnitude, thickness)
	frame.Position = UDim2.fromOffset((from.X + to.X) / 2, (from.Y + to.Y) / 2)
	frame.Rotation = math.deg(math.atan2(diff.Y, diff.X))
	frame.BackgroundColor3 = color
end

RunService.RenderStepped:Connect(function()
	local myRoot = player.Character and player.Character:FindFirstChild("HumanoidRootPart")
	local anyOn = State.ESPBox or State.ESPChams or State.ESPDistance or State.ESPName or State.ESPLines or State.ESPSkeleton

	for _, plr in Players:GetPlayers() do
		if plr ~= player then
			local char = plr.Character
			local hum = char and char:FindFirstChildOfClass("Humanoid")
			local root = char and char:FindFirstChild("HumanoidRootPart")
			local head = char and char:FindFirstChild("Head")

			if anyOn and hum and hum.Health > 0 and root and head then
				local d = espData[plr]
				if d and d.char ~= char then
					destroyESP(plr)
					d = nil
				end
				d = d or createESP(plr, char, root, head)

				local color = ROLE_COLORS[getRole(plr)]

				d.box.Enabled = State.ESPBox
				d.stroke.Color = color

				d.highlight.Enabled = State.ESPChams
				d.highlight.FillColor = color
				d.highlight.OutlineColor = color

				local cam = Workspace.CurrentCamera
				local drawn = false
				if State.ESPLines and cam then
					local pos, onScreen = cam:WorldToViewportPoint(root.Position)
					if onScreen then
						local vs = cam.ViewportSize
						local from = Vector2.new(vs.X / 2, vs.Y)
						local to = Vector2.new(pos.X, pos.Y)
						local diff = to - from
						d.line.Size = UDim2.fromOffset(diff.Magnitude, 1.5)
						d.line.Position = UDim2.fromOffset((from.X + to.X) / 2, (from.Y + to.Y) / 2)
						d.line.Rotation = math.deg(math.atan2(diff.Y, diff.X))
						d.line.BackgroundColor3 = color
						drawn = true
					end
				end
				d.line.Visible = drawn

				-- skeleton
				local boneList = nil
				if State.ESPSkeleton and cam then
					boneList = char:FindFirstChild("UpperTorso") and BONES_R15 or BONES_R6
				end
				local count = boneList and #boneList or 0
				for i = 1, math.max(count, d.boneMax) do
					local f = d.bones[i]
					local shown = false
					if i <= count then
						local a = char:FindFirstChild(boneList[i][1])
						local b = char:FindFirstChild(boneList[i][2])
						if a and b then
							local pa = cam:WorldToViewportPoint(a.Position)
							local pb = cam:WorldToViewportPoint(b.Position)
							if pa.Z > 0 and pb.Z > 0 then
								if not f then
									f = new("Frame", {
										AnchorPoint = Vector2.new(0.5, 0.5), BorderSizePixel = 0,
										BackgroundColor3 = color, Visible = false,
									}, espGui)
									d.bones[i] = f
									d.boneMax = math.max(d.boneMax, i)
								end
								drawLine(f, Vector2.new(pa.X, pa.Y), Vector2.new(pb.X, pb.Y), color, 1.5)
								shown = true
							end
						end
					end
					if f then f.Visible = shown end
				end

				local lines = {}
				if State.ESPName then
					table.insert(lines, plr.DisplayName)
				end
				if State.ESPDistance and myRoot then
					table.insert(lines, math.floor((myRoot.Position - root.Position).Magnitude) .. " studs")
				end
				d.text.Enabled = #lines > 0
				d.label.Text = table.concat(lines, "\n")
				d.label.TextColor3 = color
			else
				destroyESP(plr)
			end
		end
	end
end)

local function getPing()
	local ok, v = pcall(function()
		return Stats.Network.ServerStatsItem["Data Ping"]:GetValue()
	end)
	if ok and v then return math.floor(v + 0.5) end
	return math.floor(player:GetNetworkPing() * 2000)

	if now - lastTick >= 0.5 then
		local fps = math.floor(frames / (now - lastTick) + 0.5)

		pingDot.BackgroundColor3 = ping < 100 and GOOD or (ping < 200 and MID or BAD)

	end
end)

local KORBLOX_PARTS = {"RightUpperLeg", "RightLowerLeg", "RightFoot"}

local function setHeadless(char, on)
	local head = char:FindFirstChild("Head")
	if not head then return end
	head.Transparency = on and 1 or 0
	local face = head:FindFirstChildOfClass("Decal")
	if face then face.Transparency = on and 1 or 0 end
end

-- Korblox Deathspeaker look: thin dark bone-like right leg
local function setKorblox(char, on)
	for _, name in KORBLOX_PARTS do
		local part = char:FindFirstChild(name)
		if part and part:IsA("BasePart") then
			local bone = part:FindFirstChild("AstralKorblox")
			if on then
				part.Transparency = 1
				if not bone then
					bone = Instance.new("Part")
					bone.Name = "AstralKorblox"
					bone.Color = Color3.fromRGB(28, 28, 32)
					bone.Material = Enum.Material.SmoothPlastic
					bone.CanCollide = false
					bone.CanTouch = false
					bone.CanQuery = false
					bone.Massless = true
					if name == "RightFoot" then
						bone.Size = Vector3.new(0.3, 0.18, 0.8)
						bone.CFrame = part.CFrame * CFrame.new(0, -0.05, -0.15)
					else
						bone.Shape = Enum.PartType.Cylinder
						bone.Size = Vector3.new(part.Size.Y, 0.22, 0.22)
						bone.CFrame = part.CFrame * CFrame.Angles(0, 0, math.rad(90))
					end
					bone.Parent = part
					local w = Instance.new("WeldConstraint")
					w.Part0 = part
					w.Part1 = bone
					w.Parent = bone
				end
			elseif bone then
				bone:Destroy()
				part.Transparency = 0
			end
		end
	end
end

local wasHeadless, wasKorblox = false, false
RunService.RenderStepped:Connect(function()
	local char = player.Character
	if not char then return end

	if State.Headless then
		setHeadless(char, true)
	elseif wasHeadless then
		setHeadless(char, false)
	end
	wasHeadless = State.Headless

	if State.Korblox then
		setKorblox(char, true)
	elseif wasKorblox then
		setKorblox(char, false)
	end
	wasKorblox = State.Korblox
end)

-- Noclip + Anti Fling
--------------------------------------------------------------------
RunService.Stepped:Connect(function()
	local char = player.Character

	if State.Noclip and char then
		for _, o in char:GetDescendants() do
			if o:IsA("BasePart") then o.CanCollide = false end
		end
	end

	if State.AntiFling then
		-- other players become non-solid: walk through them, they can't push or fling you
		for _, plr in Players:GetPlayers() do
			local oc = plr ~= player and plr.Character
			if oc then
				for _, o in oc:GetChildren() do
					if o:IsA("BasePart") then o.CanCollide = false end
				end
			end
		end
		-- cancel abnormal velocity applied to you
		local root = char and char:FindFirstChild("HumanoidRootPart")
		if root then
			local limit = 250 + (State.SpeedOn and State.Speed or 0) + (State.JumpOn and State.Jump or 0)
			if root.AssemblyLinearVelocity.Magnitude > limit or root.AssemblyAngularVelocity.Magnitude > 60 then
				root.AssemblyLinearVelocity = Vector3.zero
				root.AssemblyAngularVelocity = Vector3.zero
			end
		end
	end
end)

local DROP_NAME = "GunDrop"

local function trackDrop(inst)
	if inst.Name == DROP_NAME then gunDrops[inst] = true end

local function clearDrop(inst)
	local d = dropData[inst]
	if d then
		d.highlight:Destroy()
		d.gui:Destroy()
		dropData[inst] = nil
	end
end

RunService.RenderStepped:Connect(function()
	local myRoot = player.Character and player.Character:FindFirstChild("HumanoidRootPart")
	for inst in gunDrops do
		if not inst:IsDescendantOf(Workspace) then
			clearDrop(inst)
			gunDrops[inst] = nil

--------------------------------------------------------------------
-- Wallbang (reworked)
-- While you hold the Gun, every wall on your aim line AND on the line
-- from you to the target stops blocking raycasts (CanQuery off). The
-- flags are reset and re-applied every frame, so only the walls that are
-- really in the way are affected and nothing stays broken.
-- Works when the gun's hit check (raycast / mouse hit) runs on the client.
--------------------------------------------------------------------
local wbDisabled = {}
local wbParams = RaycastParams.new()
wbParams.FilterType = Enum.RaycastFilterType.Exclude

local function wbUnderFeet(part, feet)
	local rel = part.CFrame:PointToObjectSpace(feet)
	local h = part.Size / 2 + Vector3.new(0.6, 0.6, 0.6)
	return math.abs(rel.X) <= h.X and math.abs(rel.Y) <= h.Y and math.abs(rel.Z) <= h.Z
end

-- disables blockers along the ray, returns where the ray ends
local function wbPenetrate(origin, direction, char, feet)
	local filter = {char}
	local endPoint = origin + direction
	for _ = 1, 14 do
		wbParams.FilterDescendantsInstances = filter
		local hit = Workspace:Raycast(origin, direction, wbParams)
		if not hit then break end
		local inst = hit.Instance
		local model = inst:FindFirstAncestorOfClass("Model")
		if model and model:FindFirstChildOfClass("Humanoid") then
			endPoint = hit.Position
			break
		end
		if wbUnderFeet(inst, feet) then
			table.insert(filter, inst) -- never remove the floor under you
		else
			inst.CanQuery = false
			wbDisabled[inst] = true
		end
	end
	return endPoint
end

setWallbang = function(on)
	if not on then
		for part in wbDisabled do
			if part.Parent then part.CanQuery = true end
		end
		table.clear(wbDisabled)
	end
end

RunService.RenderStepped:Connect(function()
	-- undo last frame
	for part in wbDisabled do
		if part.Parent then part.CanQuery = true end
	end
	table.clear(wbDisabled)

	if not State.Wallbang then return end
	local char = player.Character
	local root = char and char:FindFirstChild("HumanoidRootPart")
	local hum = char and char:FindFirstChildOfClass("Humanoid")
	local cam = Workspace.CurrentCamera
	if not (root and hum and cam and hum.Health > 0) then return end
	if not char:FindFirstChild(GUN_NAME) then return end -- Gun must be equipped

	local feet = root.Position - Vector3.new(0, hum.HipHeight + root.Size.Y / 2 + 0.5, 0)

	local vp
	if UIS.MouseEnabled and UIS.MouseBehavior ~= Enum.MouseBehavior.LockCenter then
		vp = UIS:GetMouseLocation()
	else
		vp = cam.ViewportSize / 2
	end
	local ray = cam:ViewportPointToRay(vp.X, vp.Y)

	-- 1) camera / mouse ray (what the gun script aims at)
	local endPoint = wbPenetrate(ray.Origin, ray.Direction * 1000, char, feet)

	-- 2) the shot itself: from you to that point
	local toTarget = endPoint - root.Position
	if toTarget.Magnitude > 1 then
		wbPenetrate(root.Position, toTarget, char, feet)
	end
end)

local floatRay = RaycastParams.new()
floatRay.FilterType = Enum.RaycastFilterType.Exclude
local floatLV, floatAtt, lastGround = nil, nil, nil

local function clearFloat()
	if floatLV then floatLV:Destroy() floatLV = nil end
	if floatAtt then floatAtt:Destroy() floatAtt = nil end
	lastGround = nil
end

RunService.Heartbeat:Connect(function()
	local char = player.Character
	local root = char and char:FindFirstChild("HumanoidRootPart")
	local hum = char and char:FindFirstChildOfClass("Humanoid")
	if not (root and hum) then
		clearFloat()
		return
	end

	if not State.Float then
		if floatLV then clearFloat() end
		return
	end

	-- (re)create the vertical-only velocity constraint
	if not floatLV or floatLV.Parent ~= root then
		clearFloat()
		floatAtt = Instance.new("Attachment")
		floatAtt.Name = "AstralFloatAtt"
		floatAtt.Parent = root
		floatLV = Instance.new("LinearVelocity")
		floatLV.Name = "AstralFloat"
		floatLV.Attachment0 = floatAtt
		floatLV.RelativeTo = Enum.ActuatorRelativeTo.World
		floatLV.VelocityConstraintMode = Enum.VelocityConstraintMode.Vector
		floatLV.ForceLimitMode = Enum.ForceLimitMode.PerAxis
		floatLV.MaxAxesForce = Vector3.new(0, 1e7, 0)
		floatLV.VectorVelocity = Vector3.zero
		floatLV.Parent = root
	end

	floatRay.FilterDescendantsInstances = {char}
	local hit = Workspace:Raycast(root.Position, Vector3.new(0, -500, 0), floatRay)
	if hit then lastGround = hit.Position.Y end
	if not lastGround then lastGround = root.Position.Y - hum.HipHeight - root.Size.Y / 2 end

	local target = lastGround + hum.HipHeight + root.Size.Y / 2 + State.FloatHeight
	local vy = math.clamp((target - root.Position.Y) * 10, -50, 50)
	floatLV.VectorVelocity = Vector3.new(0, vy, 0)
end)

end)

			local char = player.Character
			local root = char and char:FindFirstChild("HumanoidRootPart")
			local hum = char and char:FindFirstChildOfClass("Humanoid")
			if root and hum and hum.Health > 0 and getRole(player) == "Innocent" then
				local drop, dropPart
				for inst in gunDrops do
					if inst:IsDescendantOf(Workspace) then
						local part = inst:IsA("BasePart") and inst or inst:FindFirstChildWhichIsA("BasePart", true)
						if part then
							drop, dropPart = inst, part
							break
						end
					end
				end

				if drop then
					local cam = Workspace.CurrentCamera
					local back = root.CFrame
					local camCF, camType = cam.CFrame, cam.CameraType
					State.CamFrozen = true
					cam.CameraType = Enum.CameraType.Scriptable
					cam.CFrame = camCF

					pcall(function()
						local t0 = os.clock()
						while os.clock() - t0 < 0.4
							and State.AutoGun
							and root.Parent
							and drop:IsDescendantOf(Workspace)
							and getRole(player) == "Innocent" do
							root.CFrame = CFrame.new(dropPart.Position + Vector3.new(0, 1.5, 0))
							root.AssemblyLinearVelocity = Vector3.zero
							RunService.Heartbeat:Wait()
						end
					end)

					if root.Parent then
						root.CFrame = back
						root.AssemblyLinearVelocity = Vector3.zero
					end
					cam.CameraType = camType
					State.CamFrozen = false
					task.wait(0.5)
				end
			end
		end
	end
end)

			local strokeGoal = {Color = on and ACCENT or BORDER}
			if animated then
				tween(btn, trackGoal)
				tween(knob, knobGoal)
				tween(rowStroke, strokeGoal)
			else
				for k, v in trackGoal do btn[k] = v end
				for k, v in knobGoal do knob[k] = v end
				for k, v in strokeGoal do rowStroke[k] = v end
			end
		end
		render(false)
		btn.Activated:Connect(function()
			on = not on
			render(true)
			callback(on)
		end)
		if key then
			controls[key] = {type = "boolean", set = function(v)
				if v == on then return end
				on = v
				render(true)
				callback(v)
			end}
		end
	end

	local function addSlider(page, text, min, max, default, callback)
		local key = pendingKey
		pendingKey = nil
		local row = makeRow(page, 54)
		local lbl = rowLabel(row, text)
		lbl.Size = UDim2.new(1, -90, 0, 22)
		local valLbl = new("TextLabel", {
			Size = UDim2.fromOffset(60, 22), AnchorPoint = Vector2.new(1, 0),
			Position = UDim2.new(1, -12, 0, 6), BackgroundTransparency = 1,
			Text = tostring(default), TextColor3 = WHITE, Font = Enum.Font.GothamBold,
			TextSize = 14, TextXAlignment = Enum.TextXAlignment.Right,
		}, row)
		local bar = new("Frame", {
			Size = UDim2.new(1, -24, 0, 6), Position = UDim2.fromOffset(12, 38),
			BackgroundColor3 = OFF,
		}, row)
		corner(bar, 3)
		local a0 = (default - min) / (max - min)
		local fill = new("Frame", {
			Size = UDim2.fromScale(a0, 1), BackgroundColor3 = WHITE, BorderSizePixel = 0,
		}, bar)
		corner(fill, 3)
		local knob = new("Frame", {
			Size = UDim2.fromOffset(14, 14), AnchorPoint = Vector2.new(0.5, 0.5),
			Position = UDim2.fromScale(a0, 0.5), BackgroundColor3 = WHITE,
		}, bar)
		corner(knob, 7)
		new("UIStroke", {Thickness = 2, Color = ACCENT}, knob)
		local hit = new("TextButton", {
			Size = UDim2.new(1, 0, 0, 28), Position = UDim2.fromOffset(0, -11),
			BackgroundTransparency = 1, Text = "",
		}, bar)

		local dragging = false
		local function setFromX(x)
			local a = math.clamp((x - bar.AbsolutePosition.X) / bar.AbsoluteSize.X, 0, 1)
			local v = math.floor(min + (max - min) * a + 0.5)
			fill.Size = UDim2.fromScale(a, 1)
			knob.Position = UDim2.fromScale(a, 0.5)
			valLbl.Text = tostring(v)
			callback(v)
		end
		hit.InputBegan:Connect(function(i)
			if isPointer(i) then
				dragging = true
				page.ScrollingEnabled = false
				setFromX(i.Position.X)
			end
		end)
		UIS.InputChanged:Connect(function(i)
			if dragging and isMove(i) then setFromX(i.Position.X) end
		end)
		UIS.InputEnded:Connect(function(i)
			if dragging and isPointer(i) then
				dragging = false
				page.ScrollingEnabled = true
			end
		end)
		if key then
			controls[key] = {type = "number", set = function(v)
				v = math.clamp(math.floor(v + 0.5), min, max)
				local a = (v - min) / (max - min)
				fill.Size = UDim2.fromScale(a, 1)
				knob.Position = UDim2.fromScale(a, 0.5)
				valLbl.Text = tostring(v)
				callback(v)
			end}
		end
	end

	-- Config storage ----------------------------------------------------
	-- Uses files (writefile/readfile) when the environment has them,
	-- otherwise configs live in memory until you leave the game.
	local CFG_ROOT = "RedHubAstral"
	local CFG_DIR = CFG_ROOT .. "/configs"
	local hasFS = type(writefile) == "function" and type(readfile) == "function"
		and type(isfile) == "function" and type(listfiles) == "function"
		and type(delfile) == "function" and type(makefolder) == "function"
		and type(isfolder) == "function"
	local memStore = {}

	local function ensureDir()
		if not hasFS then return end
		pcall(function()
			if not isfolder(CFG_ROOT) then makefolder(CFG_ROOT) end
			if not isfolder(CFG_DIR) then makefolder(CFG_DIR) end
		end)
	end

	local function cleanName(n)
		n = (tostring(n or ""):gsub("[^%w _%-]", ""))
		n = n:match("^%s*(.-)%s*$") or ""
		return string.sub(n, 1, 24)
	end

	local function cfgPath(name)
		return CFG_DIR .. "/" .. name .. ".json"
	end

	local function listConfigs()
		local names = {}
		if hasFS then
			ensureDir()
			local ok, files = pcall(listfiles, CFG_DIR)
			if ok and type(files) == "table" then
				for _, f in files do
					local n = tostring(f):match("([^/\\]+)%.json$")
					if n then table.insert(names, n) end
				end
			end
		else
			for n in memStore do table.insert(names, n) end
		end
		table.sort(names, function(a, b) return a:lower() < b:lower() end)
		return names
	end

	local function findConfig(name)
		for _, n in listConfigs() do
			if n:lower() == name:lower() then return n end
		end
	end

	local function writeConfig(name, data)
		if hasFS then
			ensureDir()
			return (pcall(writefile, cfgPath(name), data))
		end
		memStore[name] = data
		return true
	end

	local function readConfig(name)
		if hasFS then
			local ok, data = pcall(function()
				if isfile(cfgPath(name)) then return readfile(cfgPath(name)) end
			end)
			return ok and data or nil
		end
		return memStore[name]
	end

	local function removeConfig(name)
		if hasFS then
			pcall(delfile, cfgPath(name))
		else
			memStore[name] = nil
		end
	end

	local function snapshot()
		local data = {}
		for key in controls do
			data[key] = State[key]
		end
		return HttpService:JSONEncode(data)
	end

	local function applyConfig(json)
		local ok, data = pcall(function() return HttpService:JSONDecode(json) end)
		if not ok or type(data) ~= "table" then return false end
		for key, val in data do
			local c = controls[key]
			if c and type(val) == c.type then
				pcall(c.set, val)
			end
		end
		return true
	end

	}, listHolder)

	local selected, delArm = nil, nil
	local refreshList

	refreshList = function()
		for _, c in listHolder:GetChildren() do
			if c:IsA("TextButton") or c:IsA("TextLabel") then c:Destroy() end
		end
		local names = listConfigs()
		if selected and not table.find(names, selected) then selected = nil end

		if #names == 0 then
			new("TextLabel", {
				Size = UDim2.new(1, 0, 0, 24), BackgroundTransparency = 1,
				Text = "No configs yet", TextColor3 = WHITE, TextTransparency = 0.4,
				Font = Enum.Font.Gotham, TextSize = 13,
			}, listHolder)
			return
		end

		for i, n in names do
			local isSel = (n == selected)
			local b = new("TextButton", {
				Size = UDim2.new(1, 0, 0, 28), LayoutOrder = i,
				BackgroundColor3 = isSel and TAB_ACTIVE or PANEL,
				Text = n, TextColor3 = WHITE, Font = Enum.Font.GothamMedium, TextSize = 13,
				AutoButtonColor = false,
			}, listHolder)
			corner(b, 7)
			local st = new("UIStroke", {Thickness = 1, Color = isSel and WHITE or BORDER}, b)
			b.Activated:Connect(function()
				selected = n
				delArm = nil
				nameBox.Text = ""
				refreshList()
				setStatus("Selected '" .. n .. "'")
			end)
		end
	end

			setStatus("'" .. existing .. "' already exists - select it and press Save")
			return
		end
		if writeConfig(name, snapshot()) then
			selected = name
			nameBox.Text = ""
			refreshList()
			setStatus("Created '" .. name .. "'")
		else
			setStatus("Could not write the config")
		end
	end)

	cfgButton("Save", 2, 4, function()
		if not selected then setStatus("Select a config first") return end
		if writeConfig(selected, snapshot()) then
			setStatus("Saved '" .. selected .. "'")
		else
			setStatus("Could not write the config")
		end
	end)

	cfgButton("Load", 3, 4, function()
		if not selected then setStatus("Select a config first") return end
		local data = readConfig(selected)
		if not data then
			setStatus("Config not found")
			refreshList()
			return
		end
		if applyConfig(data) then
			setStatus("Loaded '" .. selected .. "'")
		else
			setStatus("Config is corrupted")
		end
	end)

	cfgButton("Delete", 4, 4, function()
		if not selected then setStatus("Select a config first") return end
		local name = selected
		if delArm ~= name then
			delArm = name
			setStatus("Press Delete again to remove '" .. name .. "'")
			task.delay(3, function()
				if delArm == name then delArm = nil end
			end)
			return
		end
		delArm = nil
		removeConfig(name)
		selected = nil
		refreshList()
		setStatus("Deleted '" .. name .. "'")
	end)

	refreshList()
